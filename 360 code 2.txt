                        hotspotIdText: hotspotIdText, cardUrl: cardUrl, floorMapDetails: floorMapDetails, showCard: showCard, showHotspot: showHotspot, getPitchYaw: getPitchYaw
                    };
                }).filter(scene => scene.imageUrl && scene.isVisible && scene.id);

                // --- Process Sheet 2: Additional Hotspot Data ---
                const hotspotSheetName = "2. Hotspots Sheet";
                const hotspotWorksheet = workbook.Sheets[hotspotSheetName];
                if (!hotspotWorksheet) {
                    console.warn(`Sheet "${hotspotSheetName}" not found. No additional hotspots will be added.`); additionalHotspots = [];
                } else {
                    const hotspotJsonData = XLSX.utils.sheet_to_json(hotspotWorksheet, { defval: null });
                    console.log(`Processing Sheet 2: "${hotspotSheetName}" (${hotspotJsonData.length} rows)`);
                    const hotspotColumns = Object.keys(hotspotJsonData[0] || {});
                    const getHotspotVal = (index, row) => (hotspotColumns[index] && row[hotspotColumns[index]] !== null && row[hotspotColumns[index]] !== undefined) ? row[hotspotColumns[index]] : null;

                    additionalHotspots = hotspotJsonData.map((row, index) => {
                        const targetSceneId = getString(getHotspotVal(2, row)); // Column C
                        const name = getString(getHotspotVal(4, row), `Hotspot ${index + 1}`); // Column E
                        const pitch = parseNumeric(getHotspotVal(8, row), 0);  // Column I
                        const yaw = parseNumeric(getHotspotVal(9, row), 0);    // Column J
                        const cardImageUrl = getString(getHotspotVal(12, row)); // Column M (Info Card Image)
                        const cardTitle = getString(getHotspotVal(13, row));    // Column N (Info Card Title)
                        const cardDescription = getString(getHotspotVal(14, row)); // Column O (Info Card Desc)
                        const cardCurrency = getString(getHotspotVal(15, row)); // Column P (Info Card Currency)
                        const cardPrice = getString(getHotspotVal(16, row));    // Column Q (Info Card Price)
                        const sheet2ImageUrl = getString(getHotspotVal(17, row)); // Column R (VO Card Image)
                        const sheet2AudioSource = getString(getHotspotVal(18, row)); // Column S (VO Card Audio/Text)
                        // --- FIX: Read Column T for VO Active state ---
                        const sheet2VoActiveVal = getHotspotVal(19, row); // Column T
                        const sheet2VoActive = parseBoolean(sheet2VoActiveVal, false); // Default to false if empty/invalid

                        if (targetSceneId) {
                            return {
                                targetSceneId: targetSceneId, name: name, pitch: pitch, yaw: yaw,
                                cardImageUrl: cardImageUrl, cardTitle: cardTitle, cardDescription: cardDescription, cardCurrency: cardCurrency, cardPrice: cardPrice,
                                sheet2ImageUrl: sheet2ImageUrl,
                                sheet2AudioSource: sheet2AudioSource,
                                sheet2VoActive: sheet2VoActive // Store the flag
                            };
                        }
                        return null;
                    }).filter(hotspot => hotspot !== null);
                    console.log(`Loaded ${additionalHotspots.length} additional hotspots from Sheet 2.`);
                }

                if (scenes.length === 0) throw new Error('No valid scenes found in Excel data (Sheet 1) or all are hidden/missing ID.');

                filteredScenes = [...scenes];
                populateCategoryFilter();
                createSlideBar();
                initializeViewer();
                setupSlideBarInteractions();
                setupVcContainerInteractions(); // Sheet 1 VC Card
                setupSheet2VoiceoverInteractions(); // Sheet 2 VO Card
                updateSlideCounter();
                animateScroll();
                startTypingAnimation();
                setupStartButton();
                setupRightCardCloseButton(); // Info Card (Right)

            } catch (error) {
                console.error('Error loading tour data:', error);
                const loadingDiv = document.getElementById('loading');
                if (loadingDiv) { loadingDiv.textContent = `Error loading tour data: ${error.message}. Check console & file link/format.`; loadingDiv.style.color = 'red'; loadingDiv.style.fontSize = '18px'; loadingDiv.style.textAlign = 'center'; loadingDiv.style.padding = '20px'; }
            }
        }

        // Initializes the Pannellum viewer
        function initializeViewer() {
            const panorama = document.getElementById('panorama');
            if (!panorama) { console.error('Panorama element not found'); document.getElementById('loading').textContent = 'Error: Panorama container not found.'; return; }

            const pannellumConfig = {
                type: 'equirectangular', scenes: {}, autoLoad: false, showControls: false, compass: false,
                draggable: true, mouseZoom: true, touchPan: true, touchZoom: true, hotSpotDebug: false, sceneFadeDuration: 1000,
            };

            try { viewer = pannellum.viewer('panorama', pannellumConfig); }
            catch (error) { console.error("Error initializing Pannellum:", error); document.getElementById('loading').textContent = 'Error initializing 360 viewer.'; return; }

            scenes.forEach((scene, index) => {
                if (!scene.id || !scene.imageUrl) { console.warn(`Skipping scene ${index}: missing ID or Image URL.`); return; }
                const sceneHotspots = [];

                // 1. Add Sheet 1 info hotspot (if enabled)
                if (scene.showHotspot) {
                    sceneHotspots.push({
                        pitch: scene.initialPitch + 5, yaw: scene.initialYaw + 10, type: 'info', text: scene.hotspotIdText || 'More Info',
                        clickHandlerFunc: handleInfoHotspotClick, clickHandlerArgs: { sceneId: scene.id, showCard: scene.showCard }
                    });
                }
                // 2. Add Sheet 2 custom hotspots
                const relatedNewHotspots = additionalHotspots.filter(hs => hs.targetSceneId === scene.id);
                relatedNewHotspots.forEach(newHotspot => {
                    sceneHotspots.push({
                        pitch: newHotspot.pitch, yaw: newHotspot.yaw, type: 'custom', cssClass: 'sheet2-hotspot',
                        createTooltipFunc: createSheet2Tooltip, createTooltipArgs: newHotspot.name,
                        clickHandlerFunc: handleSheet2HotspotClick, // Updated handler
                        clickHandlerArgs: newHotspot // Pass all hotspot data
                    });
                });

                viewer.addScene(scene.id, {
                    title: scene.name, panorama: scene.imageUrl, pitch: scene.initialPitch, yaw: scene.initialYaw, hfov: scene.initialZoom,
                    autoRotate: scene.autoRotate, hotSpots: sceneHotspots
                });
            });

            // --- Pannellum Event Listeners ---
            viewer.on('load', () => {
                const sceneId = viewer.getScene(); console.log('Viewer loaded scene:', sceneId);
                const loadingDiv = document.getElementById('loading');
                 if (sceneId) {
                     currentSceneIndex = filteredScenes.findIndex(s => s.id === sceneId);
                      if (currentSceneIndex !== -1) {
                         const currentScene = filteredScenes[currentSceneIndex];
                         currentSceneAutoRotateSpeed = currentScene.autoRotate || 0;
                         console.log(`Scene ${sceneId} loaded. BGM Active Flag: ${currentScene.bgmActive}, VC Active Flag: ${currentScene.vcActive}, AutoRotate speed: ${currentSceneAutoRotateSpeed}`);
                         updateActiveThumbnail(sceneId);
                         updateYellowHotspotTooltip();
                         playSceneAudio(sceneId); // Handles BGM/VC based on flags & stops Sheet 2 audio
                         if (tourStarted) { handleDelayedAutoRotate(sceneId); }
                         else if (currentSceneIndex === 0 && currentSceneAutoRotateSpeed !== 0) { viewer.startAutoRotate(currentSceneAutoRotateSpeed); }
                         else { viewer.stopAutoRotate(); }
                         // FIX: Reset VC card position on load unless dragged
                         if (!vcCardLastPosition.hasBeenDragged) {
                            recalculateAndSetVcCardPosition();
                         }
                      } else { console.warn(`Loaded scene ID "${sceneId}" not found in filteredScenes.`); currentSceneAutoRotateSpeed = 0; viewer.stopAutoRotate(); hideVcActorCard(); }
                 } else { console.warn("Viewer 'load' event fired but getScene() returned undefined."); currentSceneAutoRotateSpeed = 0; viewer.stopAutoRotate(); hideVcActorCard(); }
                if (loadingDiv && loadingDiv.style.display !== 'none') { loadingDiv.style.display = 'none'; }
                adjustViewerSize();
            });

            viewer.on('scenechange', (newSceneId) => {
                console.log('Scene changing to:', newSceneId);
                 clearTimeout(autoRotateTimeoutId); clearTimeout(interactionTimeoutId); viewer.stopAutoRotate();
                 // Don't hide right card on scene change, let hotspots control it
                 hideSheet2VoiceoverCard(); // Hide Sheet 2 VO card
                 stopSheet2Audio(); // Stop Sheet 2 audio/TTS

                 if (newSceneId) {
                      const newSceneIndex = filteredScenes.findIndex(s => s.id === newSceneId);
                      if (newSceneIndex !== -1) {
                           const newScene = filteredScenes[newSceneIndex]; currentSceneIndex = newSceneIndex;
                           currentSceneAutoRotateSpeed = newScene.autoRotate || 0;
                           console.log(`Scene change: New scene ${newSceneId} has autoRotate speed: ${currentSceneAutoRotateSpeed}`);
                           if (currentCard && currentCard.classList.contains('docked-card')) { // Left card docked
                               if (newScene.showCard === true) { updateCardContent(newScene); } else { hideHoverCard(currentCard); }
                           } else if (currentCard && !currentCard.classList.contains('docked-card')) { hideHoverCard(currentCard); } // Left card hovering
                           updateYellowHotspotTooltip();
                           // Restore Sheet 1 VC Actor visibility and position if needed for the new scene
                           showVcActorCardIfNeeded(newScene); // This will handle position based on right card state and drag state
                      } else {
                           console.warn(`Changing to scene ID "${newSceneId}" which was not found in filteredScenes.`); currentSceneIndex = -1; currentSceneAutoRotateSpeed = 0; if (currentCard) hideHoverCard(currentCard); updateYellowHotspotTooltip();
                           hideVcActorCard(); // Hide Sheet 1 VC if scene not found
                        }
                 } else {
                     console.warn("Scene changed but new sceneId is undefined."); currentSceneIndex = -1; currentSceneAutoRotateSpeed = 0; if (currentCard) hideHoverCard(currentCard); updateYellowHotspotTooltip();
                     hideVcActorCard(); // Hide Sheet 1 VC if scene is undefined
                    }
            });

            // Interaction Listeners for Auto-Rotate
            viewer.on('mousedown', handleInteractionStart); viewer.on('touchstart', handleInteractionStart);
            viewer.on('mouseup', handleInteractionEnd); viewer.on('touchend', handleInteractionEnd);
            viewer.on('zoomstart', handleInteractionStart); viewer.on('zoomchange', handleInteractionEnd);
            viewer.on('animatefinished', handleInteractionEnd);
            // Tooltip Update Listeners
            viewer.on('animatefinished', updateYellowHotspotTooltip); viewer.on('zoomchange', updateYellowHotspotTooltip); viewer.on('viewchange', updateYellowHotspotTooltip);

            // Load Initial Scene
             if (filteredScenes.length > 0 && filteredScenes[0]?.id) { viewer.loadScene(filteredScenes[0].id); }
             else { console.error("Cannot load initial scene: No valid scenes found after filtering."); const loadingDiv = document.getElementById('loading'); if (loadingDiv) loadingDiv.textContent = 'Error: No valid scenes found to display.'; }

            // Set Logos
            const logoImg = document.getElementById('logoImg'); const bottomLogo = document.getElementById('bottomLogo');
            if (tourLogoUrl) {
                if (logoImg) { logoImg.src = tourLogoUrl; logoImg.style.display = 'block'; logoImg.onerror = () => { console.error("Logo load error (Top Right)"); logoImg.style.display='none'; }}
                if (bottomLogo) { bottomLogo.src = tourLogoUrl; bottomLogo.style.display = 'block'; bottomLogo.onerror = () => { console.error("Logo load error (Welcome)"); bottomLogo.style.display='none'; }}
            } else { console.warn("No Logo URL found in Excel data (first row, 'logo' column)."); }

            setupAudioControls();
        }

        // --- Start Button Setup ---
        function setupStartButton() {
            const welcomeOverlay = document.getElementById('welcome-overlay'); const startButton = document.getElementById('startButton');
            if (!welcomeOverlay || !startButton || !viewer) { console.error('Welcome overlay, start button, or viewer not found'); return; }
            startButton.removeEventListener('click', handleStartTour); startButton.addEventListener('click', handleStartTour);
            function handleStartTour(e) {
                e.stopPropagation(); e.preventDefault(); if (tourStarted) return; tourStarted = true;
                welcomeOverlay.classList.add('hidden'); welcomeOverlayVisible = false; hasInteracted = true; isAudioPlaying = true; isAutoPlayEnabled = true;
                const playBtn = document.getElementById('playBtn'); if (playBtn) playBtn.classList.add('paused'); // Set button to 'paused' state (showing pause icon)
                console.log("Start Tour clicked: Interaction registered, Audio Playing set to true.");
                viewer.stopAutoRotate(); clearTimeout(autoRotateTimeoutId); clearTimeout(interactionTimeoutId);
                console.log(`Animating back to Pitch: ${initialScenePitch}, Yaw: ${initialSceneYaw}`);
                viewer.lookAt(initialScenePitch, initialSceneYaw, viewer.getHfov(), 500);
                setTimeout(() => { if (viewer && viewer.getScene()) { const currentSceneId = viewer.getScene(); playSceneAudio(currentSceneId); handleDelayedAutoRotate(currentSceneId); } }, 100);
            }
        }

        // --- Delayed Auto-Rotation Handling ---
        function handleDelayedAutoRotate(sceneId) {
            if (!viewer || !tourStarted) return; clearTimeout(autoRotateTimeoutId);
            const scene = filteredScenes.find(s => s.id === sceneId); if (!scene) { console.warn(`[handleDelayedAutoRotate] Scene not found: ${sceneId}`); currentSceneAutoRotateSpeed = 0; return; }
            if (currentSceneAutoRotateSpeed !== 0 && !isInteracting) {
                console.log(`[handleDelayedAutoRotate] Scheduling auto-rotate for scene ${sceneId} in 25s. Speed: ${currentSceneAutoRotateSpeed}`);
                autoRotateTimeoutId = setTimeout(() => { if (!isInteracting) { console.log(`[handleDelayedAutoRotate] Starting auto-rotate now for scene ${sceneId}. Speed: ${currentSceneAutoRotateSpeed}`); viewer.startAutoRotate(currentSceneAutoRotateSpeed); } else { console.log(`[handleDelayedAutoRotate] Interaction started during timeout. Aborting.`); } }, 25000);
            } else { console.log(`[handleDelayedAutoRotate] Auto-rotate not scheduled for scene ${sceneId}. Speed: ${currentSceneAutoRotateSpeed}, Interacting: ${isInteracting}`); viewer.stopAutoRotate(); }
        }

        // --- Interaction Handling for Auto-Rotate ---
        function handleInteractionStart() {
            if (!viewer || !tourStarted) return;
            if (!isInteracting) { console.log("[Interaction Start] Stopping auto-rotate and clearing timeouts."); isInteracting = true; viewer.stopAutoRotate(); clearTimeout(autoRotateTimeoutId); clearTimeout(interactionTimeoutId); }
        }
        function handleInteractionEnd() {
            if (!viewer || !tourStarted || !isInteracting) return;
            console.log("[Interaction End] Scheduling auto-rotate restart check in 25s."); isInteracting = false; clearTimeout(interactionTimeoutId);
            interactionTimeoutId = setTimeout(() => { if (!isInteracting && currentSceneAutoRotateSpeed !== 0) { console.log(`[Interaction End Timeout] Restarting auto-rotate. Speed: ${currentSceneAutoRotateSpeed}`); viewer.startAutoRotate(currentSceneAutoRotateSpeed); } else { console.log(`[Interaction End Timeout] Auto-rotate not restarting. Interacting: ${isInteracting}, Speed: ${currentSceneAutoRotateSpeed}`); } }, 25000);
        }

        // --- Category Filter and Slide Bar UI ---
        function populateCategoryFilter() {
            const categoryList = document.getElementById('categoryList'); if (!categoryList) return; categoryList.innerHTML = '';
            const fixedCategories = ['All Floors', 'Ground Floor', 'First Floor'].reverse();
            fixedCategories.forEach(category => {
                const item = document.createElement('div'); item.className = 'category-item'; item.textContent = category;
                item.dataset.category = category === 'All Floors' ? 'all' : category; if (category === 'All Floors') item.classList.add('active');
                item.addEventListener('click', (e) => {
                    e.stopPropagation(); document.querySelectorAll('.category-item').forEach(i => i.classList.remove('active')); item.classList.add('active');
                    filterScenes(item.dataset.category); toggleSlideBar(true, true); toggleCategoryList(false);
                     if (currentCard && !currentCard.classList.contains('docked-card')) { hideHoverCard(); } hideRightCard(); hideSheet2VoiceoverCard();
                });
                item.addEventListener('mouseenter', (e) => {
                    e.stopPropagation(); if (hoverTimeout) clearTimeout(hoverTimeout); const hoveredCategory = item.dataset.category;
                    if (hoveredCategory === 'all') { if (currentCard && !currentCard.classList.contains('docked-card')) { hideHoverCard(); } return; }
                    const currentActiveScene = (filteredScenes && currentSceneIndex >= 0 && currentSceneIndex < filteredScenes.length) ? filteredScenes[currentSceneIndex] : null;
                    const shouldShowCard = currentActiveScene && currentActiveScene.category === hoveredCategory && currentActiveScene.showCard === true;
                    if (shouldShowCard) { showHoverCard(currentActiveScene, true, e.target, true); }
                    else { if (currentCard && !currentCard.classList.contains('docked-card')) { hideHoverCard(); } }
                });
                item.addEventListener('mouseleave', (e) => {
                    e.stopPropagation(); if (hoverTimeout) clearTimeout(hoverTimeout);
                    if (currentCard && !currentCard.classList.contains('pinned') && !currentCard.classList.contains('docked-card')) {
                        hoverTimeout = setTimeout(() => { if (currentCard && !currentCard.classList.contains('docked-card') && !currentCard.classList.contains('pinned') && !currentCard.matches(':hover')) { hideHoverCard(); } }, 300);
                    }
                });
                categoryList.appendChild(item);
            });
        }
        function filterScenes(category) {
            const categoryName = document.getElementById('categoryName'); if (!categoryName) return;
            if (category === 'all') { filteredScenes = [...scenes]; categoryName.textContent = 'All Floors'; }
            else { filteredScenes = scenes.filter(scene => scene.category === category); categoryName.textContent = category; }
            const thumbnailContainer = document.getElementById('thumbnailContainer'); if (!thumbnailContainer) return; thumbnailContainer.innerHTML = ''; createSlideBar();
            if (filteredScenes.length > 0 && viewer) { currentSceneIndex = 0; viewer.loadScene(filteredScenes[0].id); }
            else if (viewer) { console.warn(`No scenes found for category: ${category}`); currentSceneIndex = -1; updateSlideCounter(); viewer.stopAutoRotate(); clearTimeout(autoRotateTimeoutId); clearTimeout(interactionTimeoutId); currentSceneAutoRotateSpeed = 0; }
        }
        function createSlideBar() {
            const thumbnailContainer = document.getElementById('thumbnailContainer'); if (!thumbnailContainer) return; thumbnailContainer.innerHTML = '';
            filteredScenes.forEach((scene, index) => {
                const thumbnail = document.createElement('div'); thumbnail.className = 'thumbnail'; thumbnail.dataset.sceneId = scene.id;
                const img = document.createElement('img'); img.src = scene.thumbnailUrl || scene.imageUrl; img.alt = scene.name;
                img.onerror = () => {
                    console.error(`Failed to load thumbnail for ${scene.name}: ${img.src}`);
                    const fallbackSrc = (scene.imageUrl && scene.imageUrl !== img.src) ? scene.imageUrl : null;
                    if (fallbackSrc) {
                        console.log(`Attempting fallback image: ${fallbackSrc}`);
                        img.src = fallbackSrc;
                        img.onerror = () => {
                            console.error(`Failed fallback image for ${scene.name}: ${fallbackSrc}`);
                            thumbnail.style.backgroundColor = '#555';
                            thumbnail.innerHTML = `<span style="color:white; font-size:10px; padding: 5px; text-align:center; display:block;">Load Error</span>`;
                        };
                    } else {
                        thumbnail.style.backgroundColor = '#555';
                        thumbnail.innerHTML = `<span style="color:white; font-size:10px; padding: 5px; text-align:center; display:block;">Load Error</span>`;
                    }
                };
                thumbnail.appendChild(img);
                const nameSpan = document.createElement('span'); nameSpan.className = 'thumbnail-name'; nameSpan.textContent = scene.name; thumbnail.appendChild(nameSpan);
                thumbnail.addEventListener('click', (e) => { e.stopPropagation(); if (viewer && viewer.getScene() !== scene.id) { viewer.loadScene(scene.id); hideRightCard(); hideSheet2VoiceoverCard(); } });
                thumbnailContainer.appendChild(thumbnail);
            });
            if (thumbnailContainer.children.length > 0 && viewer && viewer.getScene()) { updateActiveThumbnail(viewer.getScene()); }
            else if (thumbnailContainer.children.length > 0) { thumbnailContainer.children[0].classList.add('active'); }
            updateSlideCounter(); updateScrollIndicator();
        }
        function setupSlideBarInteractions() {
            const slideBar = document.getElementById('slideBar'); const thumbnailContainer = document.getElementById('thumbnailContainer');
            const slideArrow = document.getElementById('slideArrow'); const categoryIcon = document.getElementById('categoryIcon');
            const categoryList = document.getElementById('categoryList'); const controlBar = document.getElementById('controlBar');
            const scrollIndicator = document.getElementById('scrollIndicator'); const scrollProgress = document.getElementById('scrollProgress');
            const fullscreenBtn = document.getElementById('fullscreenBtn');
            if (!slideBar || !thumbnailContainer || !slideArrow || !categoryIcon || !categoryList || !controlBar || !scrollIndicator || !scrollProgress || !fullscreenBtn) { console.error('Missing DOM elements for slide bar interactions'); return; }

            slideBar.addEventListener('mouseenter', () => { if (!slideBarPinned) toggleSlideBar(true, false); });
            slideBar.addEventListener('mouseleave', () => { if (!slideBarPinned) toggleSlideBar(false, false); });
            slideArrow.addEventListener('click', (e) => { e.stopPropagation(); slideBarPinned = !slideBarPinned; toggleSlideBar(slideBarPinned, slideBarPinned); });
            categoryIcon.addEventListener('mousedown', (e) => { e.stopPropagation(); toggleCategoryList(!categoryListVisible); });
            document.addEventListener('mouseup', (e) => { if (categoryListVisible && !categoryList.contains(e.target) && !categoryIcon.contains(e.target)) { toggleCategoryList(false); } });
            categoryList.addEventListener('mouseenter', () => { if (categoryListVisible) toggleCategoryList(true); });
            categoryList.addEventListener('mouseleave', () => { if (categoryListVisible && !categoryIcon.matches(':hover')) toggleCategoryList(false); });

            document.addEventListener('click', (e) => {
                const hoverCardEl = document.getElementById('hoverCard'); const sheet2CardEl = document.getElementById('sheet2Card'); const sheet2VoCardEl = document.getElementById('sheet2VoiceoverCard');
                const isOutsideInteractive = !slideBar.contains(e.target) && !controlBar.contains(e.target) && !categoryList.contains(e.target) && !categoryIcon.contains(e.target) && !fullscreenBtn.contains(e.target) &&
                                             !(hoverCardEl && hoverCardEl.contains(e.target)) && !(sheet2CardEl && sheet2CardEl.contains(e.target)) && !(sheet2VoCardEl && sheet2VoCardEl.contains(e.target));
                if (isOutsideInteractive) {
                    if (slideBarPinned) { slideBarPinned = false; toggleSlideBar(false, false); }
                    if (categoryListVisible) { toggleCategoryList(false); }
                    if (currentCard && !currentCard.classList.contains('docked-card')) { hideHoverCard(); }
                }
                if (!hasInteracted) hasInteracted = true;
            });

            document.addEventListener('mousemove', (e) => {
                if (isDraggingScrollIndicator) {
                    const containerWidth = scrollIndicator.offsetWidth; const clickPosition = e.clientX - scrollIndicator.getBoundingClientRect().left;
                    const scrollPercentage = Math.min(Math.max(clickPosition / containerWidth, 0), 1); const maxScroll = thumbnailContainer.scrollWidth - thumbnailContainer.clientWidth;
                    targetScrollLeft = maxScroll * scrollPercentage; thumbnailContainer.scrollLeft = targetScrollLeft; currentScrollLeft = targetScrollLeft; updateScrollIndicator(); e.preventDefault();
                }
                 if (isDraggingVc) { // Sheet 1 VC Card Drag
                    const newRight = vcOffsetX - (e.pageX - vcStartX); const newBottom = vcOffsetY - (e.pageY - vcStartY);
                    vcActorContainerEl.style.right = `${newRight}px`; vcActorContainerEl.style.bottom = `${newBottom}px`;
                    // Store the dragged position
                    vcCardLastPosition.right = `${newRight}px`;
                    vcCardLastPosition.bottom = `${newBottom}px`;
                    vcCardLastPosition.hasBeenDragged = true; // Mark as dragged
                    e.preventDefault();
                 }
                 if (isDraggingSheet2Vo) { // Sheet 2 VO Card Drag
                    const newRight = sheet2VoOffsetX - (e.pageX - sheet2VoStartX);
                    let newTop = 'auto', newBottom = 'auto';
                    // Calculate new position based on whether it was initially top or bottom aligned
                    if (sheet2VoCardLastPosition.top !== 'auto') {
                        newTop = sheet2VoOffsetY + (e.pageY - sheet2VoStartY);
                        sheet2VoiceoverCardEl.style.top = `${newTop}px`;
                        sheet2VoiceoverCardEl.style.bottom = 'auto';
                        sheet2VoCardLastPosition.top = `${newTop}px`;
                        sheet2VoCardLastPosition.bottom = 'auto';
                    } else {
                        newBottom = sheet2VoOffsetY - (e.pageY - sheet2VoStartY);
                        sheet2VoiceoverCardEl.style.bottom = `${newBottom}px`;
                        sheet2VoiceoverCardEl.style.top = 'auto';
                        sheet2VoCardLastPosition.bottom = `${newBottom}px`;
                        sheet2VoCardLastPosition.top = 'auto';
                    }
                    sheet2VoiceoverCardEl.style.right = `${newRight}px`;
                    sheet2VoCardLastPosition.right = `${newRight}px`;
                    sheet2VoCardLastPosition.hasBeenDragged = true; // Mark as dragged
                    e.preventDefault();
                 }
            });

            document.addEventListener('mouseup', () => {
                if (isDraggingScrollIndicator) { isDraggingScrollIndicator = false; document.body.style.cursor = ''; }
                if (isDraggingVc) { isDraggingVc = false; document.body.style.cursor = ''; }
                if (isDraggingSheet2Vo) { isDraggingSheet2Vo = false; document.body.style.cursor = ''; }
            });

            scrollIndicator.addEventListener('mousedown', (e) => {
                isDraggingScrollIndicator = true; document.body.style.cursor = 'ew-resize'; const containerWidth = scrollIndicator.offsetWidth; const clickPosition = e.clientX - scrollIndicator.getBoundingClientRect().left;
                const scrollPercentage = Math.min(Math.max(clickPosition / containerWidth, 0), 1); const maxScroll = thumbnailContainer.scrollWidth - thumbnailContainer.clientWidth;
                targetScrollLeft = maxScroll * scrollPercentage; thumbnailContainer.scrollLeft = targetScrollLeft; currentScrollLeft = targetScrollLeft; updateScrollIndicator(); e.preventDefault(); e.stopPropagation();
            });

            slideBar.addEventListener('touchstart', (e) => {
                 if (e.target === scrollIndicator || scrollIndicator.contains(e.target)) return;
                touchStartX = e.touches[0].pageX; touchStartY = e.touches[0].pageY; touchStartScrollLeft = thumbnailContainer.scrollLeft; isDraggingThumbnails = true; scrollAnimationStartTime = null;
            }, { passive: false });
            slideBar.addEventListener('touchmove', (e) => {
                if (!isDraggingThumbnails) return; const touchX = e.touches[0].pageX; const touchY = e.touches[0].pageY; const xDiff = touchX - touchStartX; const yDiff = touchY - touchStartY;
                if (Math.abs(xDiff) > Math.abs(yDiff)) {
                    targetScrollLeft = touchStartScrollLeft - xDiff; const maxScroll = thumbnailContainer.scrollWidth - thumbnailContainer.clientWidth; targetScrollLeft = Math.max(0, Math.min(maxScroll, targetScrollLeft));
                    thumbnailContainer.scrollLeft = targetScrollLeft; currentScrollLeft = targetScrollLeft; updateScrollIndicator(); e.preventDefault();

        function fadeIn(audio, targetVolume, startVolume = 0) {
            if (!audio || !audio.src || (!audio.paused && audio.volume >= targetVolume)) return; // Check src
            clearInterval(audio.fadeInterval); audio.volume = startVolume;

            function startFadeInterval() {
                 // Check if audio is still paused *after* the play promise resolves, indicating potential failure
                 if (audio.paused && startVolume === 0) { console.warn("Audio play likely failed (still paused after play attempt), aborting fade-in for:", audio.src); clearInterval(audio.fadeInterval); return; }
                 const fadeDuration = 1000; const steps = 50; const stepDuration = fadeDuration / steps; const volumeStep = (targetVolume - audio.volume) / steps;
                 audio.fadeInterval = setInterval(() => {
                    // Check if audio became paused *during* the fade-in (e.g., user paused again)
                    if (audio.paused) { console.warn("Audio paused during fadeIn for:", audio.src); clearInterval(audio.fadeInterval); return; }
                    let newVolume = audio.volume + volumeStep;
                    if ((volumeStep > 0 && newVolume >= targetVolume) || (volumeStep < 0 && newVolume <= targetVolume) || newVolume < 0 || newVolume > 1) {
                        audio.volume = Math.max(0, Math.min(1, targetVolume)); // Clamp volume
                        clearInterval(audio.fadeInterval);
                    } else {
                        audio.volume = newVolume;
                    }
                 }, stepDuration);
            }
            if (audio.paused) {
                // Attempt to play and *then* start the fade interval
                 audio.play().then(startFadeInterval).catch(err => { console.error('Play Error during fadeIn:', err, 'Src:', audio.src); clearInterval(audio.fadeInterval); });
            }
            else {
                // If already playing (e.g., restoring volume), just start the fade
                startFadeInterval();
            }
        }
        // --- FIX: Ensure fadeOut reliably pauses the audio ---
        function fadeOut(audio) {
            // Check if audio element exists, has a source, and is not already paused
            if (!audio || !audio.src || audio.paused) return;
            clearInterval(audio.fadeInterval); // Clear any existing fade interval
            const fadeDuration = 500; const steps = 25; const stepDuration = fadeDuration / steps; const startVolume = audio.volume;
            // Ensure startVolume is a valid number > 0, default to 1 otherwise
            const initialVolume = (typeof startVolume === 'number' && !isNaN(startVolume) && startVolume > 0) ? startVolume : 1;
            const volumeStep = initialVolume / steps; // Calculate volume decrease per step

            audio.fadeInterval = setInterval(() => {
                // If audio somehow gets paused externally during fade, stop the interval
                if (audio.paused) { clearInterval(audio.fadeInterval); audio.volume = 0; return; }
                let newVolume = audio.volume - volumeStep;
                if (newVolume <= 0) {
                    audio.pause(); // <<< FIX: Explicitly pause the audio
                    audio.volume = 0; // Ensure volume is 0
                    clearInterval(audio.fadeInterval); // Stop the interval
                    console.log("Audio paused by fadeOut:", audio.src);
                } else {
                    audio.volume = newVolume; // Decrease volume
                }
            }, stepDuration);
        }

        function playTTS(text, sheetNumber) { // Added sheetNumber parameter (1 or 2)
            if (!window.speechSynthesis) { console.error('TTS not supported.'); return; }
            if (!text || typeof text !== 'string' || text.trim() === '') { console.warn('Empty TTS text.'); return; }

            // Cancel any ongoing speech *before* creating the new utterance
            // This is crucial to prevent overlaps if called rapidly
            if (window.speechSynthesis.speaking || window.speechSynthesis.pending) {
                 console.log("TTS: Cancelling previous speech before starting new one.");
                 window.speechSynthesis.cancel(); // This should trigger onend/onerror of the previous utterance
            }
            // Reset states immediately after cancelling, just in case onend doesn't fire reliably
            isTTSPlaying = false;
            isSheet2TTSPlaying = false;
            ttsUtterance = null; // Clear previous utterance refs
            sheet2TtsUtterance = null;

            // Use appropriate utterance object and state variable based on sheet number
            let utterance = new SpeechSynthesisUtterance(text.trim());
            if (sheetNumber === 2) { sheet2TtsUtterance = utterance; } else { ttsUtterance = utterance; }

            utterance.volume = 1; utterance.rate = 1; utterance.pitch = 1;
            const voices = window.speechSynthesis.getVoices();
            if (voices.length > 0) {
                 let selectedVoice = voices.find(v => v.lang.startsWith('en') && v.localService) || voices.find(v => v.lang.startsWith('en')) || voices.find(v => v.localService) || voices[0];
                 utterance.voice = selectedVoice;
            } else { console.warn('No TTS voices available. Using default.'); }

            utterance.onstart = () => {
                console.log(`TTS (Sheet ${sheetNumber}) started: "${text.substring(0, 30)}..."`);
                if (sheetNumber === 2) isSheet2TTSPlaying = true; else isTTSPlaying = true;
                manageDucking();
            };
            utterance.onend = () => {
                console.log(`TTS (Sheet ${sheetNumber}) ended`);
                if (sheetNumber === 2) { isSheet2TTSPlaying = false; sheet2TtsUtterance = null; }
                else { isTTSPlaying = false; ttsUtterance = null; }
                const sceneId = viewer?.getScene(); if (!sceneId) return; const scene = filteredScenes.find(s => s.id === sceneId);
                // Restore BGM only if no *other* voice is playing
                if (scene && isBgmOn && scene.bgmActive && bgmAudio.src && !bgmAudio.paused && !isTTSPlaying && !isSheet2TTSPlaying && vcAudio.paused && sheet2Audio.paused) {
                     fadeIn(bgmAudio, 1, bgmAudio.volume);
                }
            };
            utterance.onerror = (e) => {
                console.error(`TTS (Sheet ${sheetNumber}) Error:`, e.error || 'Unknown', "for text:", `"${text.substring(0, 30)}..."`);
                if (sheetNumber === 2) { isSheet2TTSPlaying = false; sheet2TtsUtterance = null; }
                else { isTTSPlaying = false; ttsUtterance = null; }
                const sceneId = viewer?.getScene(); if (!sceneId) return; const scene = filteredScenes.find(s => s.id === sceneId);
                 // Restore BGM only if no *other* voice is playing
                if (scene && isBgmOn && scene.bgmActive && bgmAudio.src && !bgmAudio.paused && !isTTSPlaying && !isSheet2TTSPlaying && vcAudio.paused && sheet2Audio.paused) {
                     fadeIn(bgmAudio, 1, bgmAudio.volume);
                 }
            };
             utterance.onpause = () => { console.log(`TTS (Sheet ${sheetNumber}) paused`); if (sheetNumber === 2) isSheet2TTSPlaying = false; else isTTSPlaying = false; };
             utterance.onresume = () => { console.log(`TTS (Sheet ${sheetNumber}) resumed`); if (sheetNumber === 2) isSheet2TTSPlaying = true; else isTTSPlaying = true; };

            console.log(`Requesting speech for TTS (Sheet ${sheetNumber}): "${text.substring(0, 30)}..."`);
            // Use a small delay to ensure cancellation completes before speaking
            setTimeout(() => {
                 // Double check synth isn't busy before speaking
                 if (window.speechSynthesis.speaking || window.speechSynthesis.pending) {
                      console.warn("Synth still busy after delay, aborting TTS speak request.");

                      isTTSPlaying = false;
                      isSheet2TTSPlaying = false;
                      if (sheetNumber === 2) sheet2TtsUtterance = null; else ttsUtterance = null;

            } else { fadeOut(bgmAudio); bgmAudio.src = ''; bgmLoaded = false; }

        }

                } else {
                    if (type === 'Sheet2VO') sheet2TtsUtterance = null; else ttsUtterance = null;
                    return false;
                }
            }
            // Handle Audio URL
            if (!audioElement) return false; const currentSrc = audioElement.src || '';

            if (currentSrc.endsWith(urlOrText)) { if (!audioElement.paused || audioElement.readyState >= 2) return true; }

            audioElement.src = urlOrText; audioElement.load(); // Set src and trigger load
             return new Promise((resolve) => {

                 const canPlayHandler = () => { if (resolved) return; resolved = true; clearTimeout(loadTimeout); console.log(`${type} canplay: ${urlOrText}`); resolve(true); };
                 const errorHandler = (e) => { if (resolved) return; resolved = true; clearTimeout(loadTimeout); console.error(`${type} Load Error:`, e.message || e, 'URL:', urlOrText); audioElement.src = ''; resolve(false); };
                 // Timeout for loading

             });

            if (currentCard && !currentCard.classList.contains('docked-card')) hideHoverCard();

            // 2. Show Right Info Card (Always happens)

            const priceEl = document.getElementById('sheet2CardPrice'); const descriptionEl = document.getElementById('sheet2CardDescription');
            if (!card || !imageEl || !titleEl || !priceEl || !descriptionEl || !hotspotData) { console.error("Cannot show right card: Missing elements or hotspot data."); return; }
            imageEl.src = hotspotData.cardImageUrl || ''; imageEl.alt = hotspotData.cardTitle || 'Product Image';
            imageEl.onerror = () => { console.error("Failed right card image load:", hotspotData.cardImageUrl); imageEl.src = 'https://placehold.co/260x200/f0f0f0/ccc?text=Image+Not+Found'; }
            titleEl.textContent = hotspotData.cardTitle || 'No Title'; descriptionEl.textContent = hotspotData.cardDescription || 'No description available.';
            const currencyRaw = hotspotData.cardCurrency || ''; const priceValue = hotspotData.cardPrice || ''; let priceHTML = '';
            const isCurrencyUrl = isValidUrl(currencyRaw);
            if (currencyRaw && isCurrencyUrl) { priceHTML = `<img src="${currencyRaw}" alt="" class="currency-symbol" onerror="this.style.display='none'; this.nextSibling.textContent = '?${priceValue}';"><span>${priceValue}</span>`; }
            else if (currencyRaw && !isCurrencyUrl) { priceHTML = `<span>${currencyRaw}${priceValue}</span>`; } else { priceHTML = `<span>${priceValue}</span>`; }
            priceEl.innerHTML = priceHTML;
            showRightCard(); // Show the info card

            // 3. Hide Sheet 1 VC Actor Card (if visible)
            hideVcActorCard();

            // 4. --- FIX: Check sheet2VoActive flag ---
            if (hotspotData.sheet2VoActive === true) {
                console.log("Sheet 2 VO is active for this hotspot (Column T is TRUE).");
                // Show Sheet 2 Voiceover Card (if image URL is valid) & Play Audio/TTS
                const voImageUrl = hotspotData.sheet2ImageUrl;
                const voAudioSource = hotspotData.sheet2AudioSource;
                showSheet2VoiceoverCard(voImageUrl, 'bottom-right'); // Show the VO card (only if image is valid)
                playSheet2Audio(voAudioSource); // Play the associated audio/TTS (this function now stops Sheet 1 VC/TTS)
            } else {
                console.log("Sheet 2 VO is inactive for this hotspot (Column T is FALSE or empty).");
                // Ensure Sheet 2 VO card is hidden and audio is stopped (already handled by scene change/stopSheet2Audio, but good to be explicit)
                hideSheet2VoiceoverCard();
                stopSheet2Audio();
            }
        }

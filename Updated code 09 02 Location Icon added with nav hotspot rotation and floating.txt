        function calculateDistance(pitch1, yaw1, pitch2, yaw2) { const toRad = deg => deg * (Math.PI / 180); const rPitch1 = toRad(pitch1), rYaw1 = toRad(yaw1), rPitch2 = toRad(pitch2), rYaw2 = toRad(yaw2); const dPitch = rPitch2 - rPitch1; let dYaw = rYaw2 - rYaw1; if (dYaw > Math.PI) dYaw -= 2 * Math.PI; if (dYaw < -Math.PI) dYaw += 2 * Math.PI; return Math.sqrt(dPitch ** 2 + (dYaw * Math.cos((rPitch1 + rYaw2) / 2)) ** 2); }
